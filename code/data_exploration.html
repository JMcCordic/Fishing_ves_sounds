Merge and reshape tables

For sequence analysis, we want (1) a wide df with one row per sequence ID with events (A/B) going across the columns and (2) a df with sequence ID and category (fishing/transit)

Sequence ID = each line & behavior combo Event = 2 states (A, B – alternating per selection) Class = 2 states (fishing, transit – determined during field trial)

# first we need all the data together to map the categories onto the sub-selections

selns_data <- selns_int_og |>

Cost matrix

  • Use standard cost matrix values: insertion/deletion cost = 1.0 missingness cost = 0.5*[highest cost in matrix] substitution cost = constant
costmatrix <- seqsubm(event_seq, 
                      method="CONSTANT",
                      cval = 2, 
                      with.missing=TRUE,
                      miss.cost=1, 
                      time.varying=FALSE, 
                      weighted=TRUE,
                      transition="both", 
                      lag=1)

hilo_cat <- intervals_cat |>
  # filter out off effort for now
  filter(activity != "Off_eff_transit") |>
  # assign new sequence IDs by fishing activity
  mutate(seq_ID_fish = case_when(grepl("dredge", seq_ID) ==TRUE ~ paste0(as.character(Line_num),"_","fishing"),
                                 TRUE ~ paste0(as.character(Line_num),"_","transit"))
  )|>
  # summarize by sequence to get median metrics across whole sequence
  group_by(seq_ID_fish) |>
  mutate(med_BB = median(BB, na.rm = TRUE),
         med_tol2k = median(TOL_2000, na.rm = TRUE)) |>
  # grop by selection to get median metrics per selection
  group_by(Selection) |>
  mutate(med_sel_BB = median(BB, na.rm = TRUE),
         med_sel_tol2k = median(TOL_2000, na.rm = TRUE),
         diff_BB = med_sel_BB - med_BB,
         diff_tol2k = med_sel_tol2k - med_tol2k, 
         event_hilo = case_when(
           diff_tol2k > 0 ~ "high",
           diff_tol2k < 0 ~ "low")
         )



hilo_seq <- hilo_cat |>
  # get rid of rows without metrics info
