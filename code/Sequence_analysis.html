Merge and reshape tables

For sequence analysis, we want (1) a wide df with one row per sequence ID with events (A/B) going across the columns and (2) a df with sequence ID and category (fishing/transit)

Sequence ID = each line & behavior combo Event = 2 states (A, B – alternating by selection) Class = 2 states (fishing, transit – determined during field trial)

# first we need all the data together to map the categories onto the sub-selections

selns_data <- selns_int_og |>
  mutate(datetime = as_datetime(`Begin Date Time`),
         # assign alternating event codes
         event_ab = rep_len(c("a","b"), length.out = n()))

# pull behaviors & GPS from metrics_og
behav_gps <- metrics_og |>
  select(datetime,Line_num, activity,lat,lon, GPS_speed_kt, Dist_slant)

metrics_join <- all_metrics_og |>
  rename("datetime" = "RLDateTimeHyd") |>
  full_join(behav_gps, by = "datetime", keep = FALSE)


# map categories from metrics df onto selections
intervals_cat <- selns_data |>
  # join dfs together
  full_join(metrics_join) |>
  # get rid of some unused columns
  select(-c(View, Channel,`Low Freq (Hz)`, `High Freq (Hz)` ,`Begin Time (s)`, `End Time (s)`, `Begin Date Time`, OL_Hz:OL_16000, TOL_Hz, TOL_12500:TOL_20000))|>
  # sort by datetime
  arrange(datetime)|>
  
  # fill in data from metrics df into selection rows
  # fill selection number down to assign metrics rows to selections
  fill(c(Selection,event_ab, `Dur 90% (s)`,Line_num, activity,lat,lon, GPS_speed_kt, Dist_slant))|>
  # create sequence identifier of line + activity
  mutate(seq_ID = case_when(activity == "Off_eff_transit" ~ paste(Line_num, activity, sep = "_"),
                            grepl("dredge", activity)==TRUE ~ paste(Line_num, "fishing", sep = "_"),
                            TRUE ~ paste(Line_num, "transit", sep = "_"))) |>
  # filter to just get times included in selection table
  filter(seq_ID != "NA_transit")|>
  group_by(Selection) |>
  mutate(sel_dist = median(round(Dist_slant,0)), 
         sel_dist_cat = case_when(round(sel_dist,0) <= 250 ~ "0-250m",
                                  round(sel_dist,0) > 250 & round(sel_dist,0) <=500 ~ "250-500m",
                                  round(sel_dist,0) > 500 & round(sel_dist,0) <=1000 ~ "500-1000m",
                                  round(sel_dist,0) > 1000 ~ ">1000m"))

# Assign sequence events based on relative SPL (high vs low) and Duration

# get 50th percentile to use as "short" vs "long" cutoff
dur_cut_val <- quantile(unique(intervals_cat$`Dur 90% (s)`))[3]

# start with intervals_cat
hilo_cat <- intervals_cat |>
  # group by seq_ID & selection to get median metrics per selection
  group_by(seq_ID, Selection) |>
  summarize(med_sel_BB = median(BB, na.rm = TRUE),
            med_sel_tol2k = median(TOL_2000, na.rm = TRUE),
            Dur_90_perc = mean(`Dur 90% (s)`)) |>
  
  # assign categories based on TOL_2000 and duration
  mutate(
    n_selns = n(), 
    
    # get lag and lead values for differences
    lag_tol2k = lag(med_sel_tol2k),
    lead_tol2k = lead(med_sel_tol2k),
    diff_tol2k = case_when(
      # if lagged value is NA (AKA first value in group), use lead value to fill it in
      is.na(lag_tol2k) ~ med_sel_tol2k - lead_tol2k,
      
      TRUE ~ med_sel_tol2k - lag_tol2k),
    # assign high/low categories based on TOL values per selection within each sequence
    tol2k_hilo = case_when(
      # For single instance of one selection for whole sequence (Sel 25, Line 4, transit),
      #   compare to previous selection (Sel 24) which is lower, so assign manually as "high"
      n_selns == 1 ~ "high",
      diff_tol2k > 0 ~ "high",
      diff_tol2k < 0 ~ "low"
    ),
    
    # assign long/short categories based on 90% duration per selection within each sequence
    dur_long_short = case_when(
      Dur_90_perc <= dur_cut_val ~ "short",
      Dur_90_perc > dur_cut_val ~ "long"
    ),
    
    # paste SPL and duration combos together (4 possible values)
    event_hilo = paste0(tol2k_hilo, "_", dur_long_short))

## ℹ Summaries were computed grouped by seq_ID and Selection.
## ℹ Output is grouped by seq_ID.
## ℹ Use `summarise(.groups = "drop_last")` to silence this message.
## ℹ Use `summarise(.by = c(seq_ID, Selection))` for per-operation grouping
##   (`?dplyr::dplyr_by`) instead.
# set up df for sequence analysis  
selns_seq <- hilo_cat |>
  # filter out off effort for now
  filter(seq_ID != "Off_eff_transit") |>
  ungroup()|>
  group_by(seq_ID) |>
  # only keep columns needed for sequence analysis
  select(seq_ID,
         event_hilo) |>
 # create time index variable per sequence
  mutate(time_idx = seq(1:n())) 


# create key for sequence IDs to fishing vs. transit
fish_states <- selns_seq |>
  select(seq_ID) |>
  mutate(fish_state = case_when(grepl("fishing", seq_ID)==TRUE ~ "fishing",
                   TRUE ~ "transit"))

Cost matrix

  • Use standard cost matrix values: insertion/deletion cost = 1.0 missingness cost = 0.5*[highest cost in matrix] substitution cost = constant
costmatrix <- seqsubm(event_seq, 
                      method="CONSTANT",
                      cval = 2, 
                      with.missing=FALSE,
                      miss.cost=1, 
                      time.varying=FALSE, 
                      weighted=TRUE,
                      transition="both", 
                      lag=1)
